cmake_minimum_required(VERSION 3.16)
project(OpenSSLClosureProbe LANGUAGES CXX)
# The shared mode is a diagnostic control, never static-build acceptance.
option(PROBE_STATIC_OPENSSL "Require OpenSSL archives" ON)
set(OPENSSL_USE_STATIC_LIBS ${PROBE_STATIC_OPENSSL})
find_package(OpenSSL REQUIRED COMPONENTS Crypto SSL)
if(PROBE_STATIC_OPENSSL)
  foreach(component CRYPTO SSL)
    if(NOT OPENSSL_${component}_LIBRARY MATCHES "\\.(a|lib)$")
      message(FATAL_ERROR "Expected archive: ${OPENSSL_${component}_LIBRARY}")
    endif()
  endforeach()
endif()
message(STATUS "OpenSSL headers: ${OPENSSL_INCLUDE_DIR}; version: ${OPENSSL_VERSION}")
message(STATUS "Crypto: ${OPENSSL_CRYPTO_LIBRARY}; SSL: ${OPENSSL_SSL_LIBRARY}")
add_executable(openssl-probe main.cpp)
target_compile_features(openssl-probe PRIVATE cxx_std_20)
target_link_libraries(openssl-probe PRIVATE OpenSSL::SSL OpenSSL::Crypto)
enable_testing()
add_test(NAME openssl-api-contract COMMAND openssl-probe)
