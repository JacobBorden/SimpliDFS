cmake_minimum_required(VERSION 3.10)
project(StaticBoostProbe LANGUAGES CXX)
set(Boost_USE_STATIC_LIBS ON)
find_package(Boost 1.86.0 EXACT REQUIRED COMPONENTS system)
get_target_property(archive Boost::system IMPORTED_LOCATION_RELEASE)
if(NOT archive MATCHES "\\.a$")
  message(FATAL_ERROR "Expected static archive, got ${archive}")
endif()
message(STATUS "Verified archive: ${archive}")
add_executable(probe main.cpp)
target_link_libraries(probe PRIVATE Boost::system)
enable_testing()
add_test(NAME static-boost-system COMMAND probe)
